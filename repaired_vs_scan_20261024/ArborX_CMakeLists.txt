cmake_minimum_required(VERSION 3.22)
project(repaired_gts_native_arborx LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(ARBORX_ENABLE_MPI OFF CACHE BOOL "" FORCE)
set(ARBORX_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(ARBORX_ENABLE_BENCHMARKS OFF CACHE BOOL "" FORCE)
add_subdirectory("${ARBORX_SOURCE}" native_arborx)
foreach(mode IN ITEMS 0 1)
  if(mode EQUAL 0)
    set(name arborx_bench)
  else()
    set(name arborx_count)
  endif()
  if(mode EQUAL 1 AND DEFINED COUNTER_SOURCE)
    add_executable(${name} "${COUNTER_SOURCE}")
  else()
    add_executable(${name} arborx_bench.cpp)
  endif()
  target_compile_definitions(${name} PRIVATE INSTRUMENT=${mode})
  target_compile_options(${name} PRIVATE -O2 -lineinfo)
  target_link_libraries(${name} PRIVATE ArborX::ArborX)
endforeach()
